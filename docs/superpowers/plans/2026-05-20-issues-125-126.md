## Task 2: Test passive capture regex patterns (Issue #125 area 4)

**Files:**
- Create: `test/passive-capture-patterns.test.js`

The `DECISION_PATTERNS` in `extensions/memory-layer/hooks/passive-capture.ts` now require contextual words ("because", "since", etc.) to reduce false positives. Tests verify both matching (true positives) and rejection (true negatives).

- [ ] **Step 1: Write the test file**

Create `test/passive-capture-patterns.test.js`:

```javascript
import { describe, it, expect } from 'vitest';

// Replicate the exact patterns from passive-capture.ts
const DECISION_PATTERNS = [
  {
    regex: /\b(I['']ll use|let's use|we should use|going with|switching to|using .* instead of)\b.*\b(because|since|reason|to avoid|for better)\b/i,
    type: 'decision',
    label: 'Design decision',
  },
  { regex: /\b(approach|strategy|architecture|pattern|design):\s.*\b(implement|chose|selected|decided)\b/i, type: 'decision', label: 'Architecture choice' },
  {
    regex: /\b(root cause|the bug was|issue is that|fixed by|workaround is to)\b/i,
    type: 'bugfix',
    label: 'Bug fix',
  },
  {
    regex: /\b(I discovered that|turns out the reason|found that .* because|note that .* limitation)\b/i,
    type: 'discovery',
    label: 'Discovery',
  },
  {
    regex: /\b(cannot .* because|constraint is|requirement is that|limitation:)\b/i,
    type: 'architecture',
    label: 'Constraint identified',
  },
];

function matchPattern(text) {
  for (const pattern of DECISION_PATTERNS) {
    if (pattern.regex.test(text)) {
      return pattern;
    }
  }
  return null;
}

describe('passive capture patterns', () => {
  describe('true positives — should match', () => {
    it('should match design decision with "because"', () => {
      const result = matchPattern("I'll use SQLite because it's simpler for local storage");
      expect(result).not.toBeNull();
      expect(result.type).toBe('decision');
    });

    it('should match "going with" + "since"', () => {
      const result = matchPattern("Going with the caching approach since it reduces latency");
      expect(result).not.toBeNull();
      expect(result.type).toBe('decision');
    });

    it('should match "switching to" + "to avoid"', () => {
      const result = matchPattern("Switching to Bun to avoid the Node.js memory issue");
      expect(result).not.toBeNull();
      expect(result.type).toBe('decision');
    });

    it('should match "using X instead of Y" + "for better"', () => {
      const result = matchPattern("Using SQLite instead of PostgreSQL for better local dev experience");
      expect(result).not.toBeNull();
      expect(result.type).toBe('decision');
    });

    it('should match architecture choice', () => {
      const result = matchPattern("approach: we decided to implement event sourcing");
      expect(result).not.toBeNull();
      expect(result.type).toBe('decision');
    });

    it('should match bug fix patterns', () => {
      expect(matchPattern("The root cause was a missing null check").type).toBe('bugfix');
      expect(matchPattern("The bug was in the parser").type).toBe('bugfix');
      expect(matchPattern("Issue is that the config was cached").type).toBe('bugfix');
      expect(matchPattern("Fixed by adding a guard clause").type).toBe('bugfix');
      expect(matchPattern("Workaround is to clear the cache").type).toBe('bugfix');
    });

    it('should match discovery patterns', () => {
      expect(matchPattern("I discovered that the cache never expires").type).toBe('discovery');
      expect(matchPattern("Turns out the reason was a race condition").type).toBe('discovery');
    });

    it('should match constraint patterns', () => {
      expect(matchPattern("Cannot use SQLite because of concurrency limits").type).toBe('architecture');
      expect(matchPattern("Constraint is that we need offline support").type).toBe('architecture');
      expect(matchPattern("Requirement is that all data stays local").type).toBe('architecture');
    });
  });

  describe('true negatives — should NOT match', () => {
    it('should not match casual "I\'ll use" without context word', () => {
      const result = matchPattern("I'll use the standard library for this task");
      expect(result).toBeNull();
    });

    it('should not match bare "going with" without reasoning', () => {
      const result = matchPattern("Going with option A");
      expect(result).toBeNull();
    });

    it('should not match random text', () => {
      expect(matchPattern("Here is the code for the function")).toBeNull();
      expect(matchPattern("Let me check the file structure")).toBeNull();
      expect(matchPattern("The test is passing now")).toBeNull();
    });

    it('should not match short messages under threshold', () => {
      // These are too short to be meaningful decisions — the extension also checks
      // text.length < 100, but the regex itself should still not match most short text
      expect(matchPattern("OK")).toBeNull();
      expect(matchPattern("Looks good")).toBeNull();
    });

    it('should not match "using" without "instead of" and context word', () => {
      const result = matchPattern("Using the helper function to parse the data");
      expect(result).toBeNull();
    });
  });
});
```

    it('should run compact at session end when runCompact is provided', () => {
      const sqlJson = vi.fn();
      const sqlRun = vi.fn();
      const jsonErrNoExit = vi.fn((msg) => ({ error: msg }));
      const trustRecovery = vi.fn(() => ({ ok: true }));
      const runCompact = vi.fn(() => ({ ok: true, pruned: 3 }));
      const result = sessionEnd(
        { sqlJson, sqlRun, jsonErrNoExit, trustRecovery, runCompact },
        { id: '10', memories: '5' },
      );
      expect(runCompact).toHaveBeenCalled();
      expect(result.compacted).toEqual({ ok: true, pruned: 3 });
    });

    it('should not fail when runCompact is not provided', () => {
      const sqlJson = vi.fn();
      const sqlRun = vi.fn();
      const jsonErrNoExit = vi.fn((msg) => ({ error: msg }));
      const trustRecovery = vi.fn(() => ({ ok: true }));
      const result = sessionEnd(
        { sqlJson, sqlRun, jsonErrNoExit, trustRecovery },
        { id: '10', memories: '5' },
      );
      expect(result.ok).toBe(true);
      expect(result.compacted).toBeNull();
    });
